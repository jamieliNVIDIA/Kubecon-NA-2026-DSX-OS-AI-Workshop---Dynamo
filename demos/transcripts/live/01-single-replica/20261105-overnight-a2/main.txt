$ cat deployment-excerpt.yaml
# deployment.yaml excerpt; image pins, storage and setup omitted.
kind: DynamoGraphDeployment
spec:
  components:
  - name: Frontend
    type: frontend
    replicas: 1
  - name: agg
    type: worker
    replicas: 1
    podTemplate:
      spec:
        containers:
        - command: [python3, -m, dynamo.vllm]
          args:
          - --model
          - nvidia/NVIDIA-Nemotron-3.5-Super-VL-120B-A12B-NVFP4
          - --tensor-parallel-size
          - '2'
          resources:
            claims: [{name: gpu-main}]
        resourceClaims:
        - name: gpu-main
          resourceClaimTemplateName: demo-inference-gpus

---
# gpu-claim-template.yaml excerpt
kind: ResourceClaimTemplate
metadata: {name: demo-inference-gpus}
spec:
  spec:
    devices:
      requests:
      - name: gpu
        exactly: {deviceClassName: gpu.nvidia.com, count: 2}
